69 /** \brief get file size */
70 unsigned int
71 getFileSize ()
72 {
73 unsigned int output = 0;
74 char *str = file_size;
75 for (int i = 0; i < 11; i++)
76 {
77 output = output * 8 + *str - '0';
78 str++;
79 }
80 return (output);
81 }
