495 /** \brief A point structure representing an N-D histogram.

499 struct Histogram
500 {

505 template <int N> std::ostream&
506 operator << (std::ostream& os, const Histogram<N>& p)
507 {
508 // make constexpr
509 PCL_IF_CONSTEXPR(N > 0)
510 {
511 os << "(" << p.histogram[0];
512 std::for_each(p.histogram + 1, std::end(p.histogram),
513 [&os](const auto& hist) { os << ", " << hist; });
514 os << ")";
515 }
