307 /** \brief A structure representing RGB color information.
308 *
309 * The RGBA information is available either as separate r, g, b, or as a
310 * packed std::uint32_t rgba value. To pack it, use:
311 *
312 * \code
313 * int rgb = ((int)r) << 16 | ((int)g) << 8 | ((int)b);
314 * \endcode
315 *
316 * To unpack it use:
317 *
318 * \code
319 * int rgb = ...;
320 * std::uint8_t r = (rgb >> 16) & 0x0000ff;
321 * std::uint8_t g = (rgb >> 8) & 0x0000ff;
322 * std::uint8_t b = (rgb) & 0x0000ff;
323 * \endcode
324 *
325 */

327 {
328 inline constexpr RGB (const _RGB &p) : RGB{p.r, p.g, p.b, p.a} {}

331

526 /** \brief A point structure representing Euclidean xyz coordinates, and the RGB color.
527 *
528 * Due to historical reasons (PCL was first developed as a ROS package), the
529 * RGB information is packed into an integer and casted to a float. This is
530 * something we wish to remove in the near future, but in the meantime, the
531 * following code snippet should help you pack and unpack RGB colors in your
532 * PointXYZRGB structure:
533 *
534 * \code
535 * // pack r/g/b into rgb
536 * std::uint8_t r = 255, g = 0, b = 0; // Example: Red color
537 * std::uint32_t rgb = ((std::uint32_t)r << 16 | (std::uint32_t)g << 8 | (std::uint32_t)b);
538 * p.rgb = *reinterpret_cast<float*>(&rgb);
539 * \endcode
540 *
541 * To unpack the data into separate values, use:
542 *
543 * \code
544 * PointXYZRGB p;
545 * // unpack rgb into r/g/b
546 * std::uint32_t rgb = *reinterpret_cast<int*>(&p.rgb);
547 * std::uint8_t r = (rgb >> 16) & 0x0000ff;
548 * std::uint8_t g = (rgb >> 8) & 0x0000ff;
549 * std::uint8_t b = (rgb) & 0x0000ff;
550 * \endcode
551 *
552 *
553 * Alternatively, from 1.1.0 onwards, you can use p.r, p.g, and p.b directly.
554 *
555 * \ingroup common
556 */

562

567 PointXYZRGB (_x, _y, _z, 0, 0, 0) {}
568
