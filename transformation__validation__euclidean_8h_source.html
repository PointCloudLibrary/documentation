50/** \brief TransformationValidationEuclidean computes an L2SQR norm between a source and
51 * target dataset.
52 *
53 * To prevent points with bad correspondences to contribute to the overall score, the
54 * class also accepts a maximum_range parameter given via \ref setMaxRange that is used
55 * as a cutoff value for nearest neighbor distance comparisons.
56 *
57 * The output score is normalized with respect to the number of valid correspondences
58 * found.
59 *
60 * Usage example:
61 * \code
62 * pcl::TransformationValidationEuclidean<pcl::PointXYZ, pcl::PointXYZ> tve;
63 * tve.setMaxRange (0.01); // 1cm
64 * double score = tve.validateTransformation (source, target, transformation);
65 * \endcode
66 *
67 * \note The class is templated on the source and target point types as well as on the
68 * output scalar of the transformation matrix (i.e., float or double). Default: float.
69 * \author Radu B. Rusu
70 * \ingroup registration
71 */
72template <typename PointSource, typename PointTarget, typename Scalar = float>
73class TransformationValidationEuclidean {
74public:
75 using Matrix4 =
76 typename TransformationValidation<PointSource, PointTarget, Scalar>::Matrix4;
77
78 using Ptr = shared_ptr<TransformationValidation<PointSource, PointTarget, Scalar>>;
79 using ConstPtr =
80 shared_ptr<const TransformationValidation<PointSource, PointTarget, Scalar>>;
81
82 using KdTree = pcl::search::Search<PointTarget>;
83 using KdTreePtr = typename KdTree::Ptr;
84
85 using PointRepresentationConstPtr =
86 typename PointRepresentation<PointTarget>::ConstPtr;
87
88 using PointCloudSourceConstPtr =
89 typename TransformationValidation<PointSource,
90 PointTarget>::PointCloudSourceConstPtr;
91 using PointCloudTargetConstPtr =
92 typename TransformationValidation<PointSource,
93 PointTarget>::PointCloudTargetConstPtr;
94
95 /** \brief Constructor.
96 * Sets the \a max_range parameter to double::max, \a threshold_ to NaN
97 * and initializes the internal search \a tree to a FLANN kd-tree.
98 */
99 TransformationValidationEuclidean()
100 : max_range_(std::numeric_limits<double>::max())
101 , threshold_(std::numeric_limits<double>::quiet_NaN())
102 {}
103
104 virtual ~TransformationValidationEuclidean() = default;
105
106 /** \brief Set the maximum allowable distance between a point and its correspondence
107 * in the target in order for a correspondence to be considered \a valid. Default:
108 * double::max. \param[in] max_range the new maximum allowable distance
109 */
110 inline void
111 setMaxRange(double max_range)
112 {
113 max_range_ = max_range;
114 }
115
116 /** \brief Get the maximum allowable distance between a point and its
117 * correspondence, as set by the user.
118 */
119 inline double
120 getMaxRange()
121 {
122 return (max_range_);
123 }
124
125 /** \brief Provide a pointer to the search object used to find correspondences in
126 * the target cloud.
127 * \param[in] tree a pointer to the spatial search object.
128 * \param[in] force_no_recompute If set to true, this tree will NEVER be
129 * recomputed, regardless of calls to setInputTarget. Only use if you are
130 * confident that the tree will be set correctly.
131 */
132 inline void
133 setSearchMethodTarget(const KdTreePtr& tree, bool force_no_recompute = false)
134 {
135 tree_ = tree;
136 force_no_recompute_ = force_no_recompute;
137 }
138
139 /** \brief Set a threshold for which a specific transformation is considered valid.
140 *
141 * \note Since we're using MSE (Mean Squared Error) as a metric, the threshold
142 * represents the mean Euclidean distance threshold over all nearest neighbors
143 * up to max_range.
144 *
145 * \param[in] threshold the threshold for which a transformation is vali
146 */
147 inline void
148 setThreshold(double threshold)
149 {
150 threshold_ = threshold;
151 }
152
153 /** \brief Get the threshold for which a specific transformation is valid. */
154 inline double
155 getThreshold()
156 {
157 return (threshold_);
158 }
159
160 /** \brief Validate the given transformation with respect to the input cloud data, and
161 * return a score.
162 *
163 * \param[in] cloud_src the source point cloud dataset
164 * \param[in] cloud_tgt the target point cloud dataset
165 * \param[out] transformation_matrix the resultant transformation matrix
166 *
167 * \return the score or confidence measure for the given
168 * transformation_matrix with respect to the input data
169 */
170 double
171 validateTransformation(const PointCloudSourceConstPtr& cloud_src,
172 const PointCloudTargetConstPtr& cloud_tgt,
173 const Matrix4& transformation_matrix) const;
174
175 /** \brief Comparator function for deciding which score is better after running the
176 * validation on multiple transforms.
177 *
178 * \param[in] score1 the first value
179 * \param[in] score2 the second value
180 *
181 * \return true if score1 is better than score2
182 */
183 virtual bool
184 operator()(const double& score1, const double& score2) const
185 {
186 return (score1 < score2);
187 }
188
189 /** \brief Check if the score is valid for a specific transformation.
190 *
191 * \param[in] cloud_src the source point cloud dataset
192 * \param[in] cloud_tgt the target point cloud dataset
193 * \param[out] transformation_matrix the transformation matrix
194 *
195 * \return true if the transformation is valid, false otherwise.
196 */
197 virtual bool
198 isValid(const PointCloudSourceConstPtr& cloud_src,
199 const PointCloudTargetConstPtr& cloud_tgt,
200 const Matrix4& transformation_matrix) const
201 {
202 if (std::isnan(threshold_)) {
203 PCL_ERROR("[pcl::TransformationValidationEuclidean::isValid] Threshold not set! "
204 "Please use setThreshold () before continuing.\n");
205 return (false);
206 }
207
208 return (validateTransformation(cloud_src, cloud_tgt, transformation_matrix) <
209 threshold_);
210 }
211
212protected:
213 /** \brief The maximum allowable distance between a point and its correspondence in
214 * the target in order for a correspondence to be considered \a valid. Default:
215 * double::max.
216 */
217 double max_range_;
218
219 /** \brief The threshold for which a specific transformation is valid.
220 * Set to NaN by default, as we must require the user to set it.
221 */
222 double threshold_;
223
224 /** \brief A pointer to the spatial search object. */
225 KdTreePtr tree_;
226
227 /** \brief A flag which, if set, means the tree operating on the target cloud
228 * will never be recomputed*/
229 bool force_no_recompute_{false};

231 /** \brief Internal point representation uses only 3D coordinates for L2 */
232 class MyPointRepresentation : public pcl::PointRepresentation<PointTarget> {
233 using pcl::PointRepresentation<PointTarget>::nr_dimensions_;
234 using pcl::PointRepresentation<PointTarget>::trivial_;
235
236 public:
237 using Ptr = shared_ptr<MyPointRepresentation>;
238 using ConstPtr = shared_ptr<const MyPointRepresentation>;
239
240 MyPointRepresentation()
241 {
242 nr_dimensions_ = 3;
243 trivial_ = true;
244 }
245
246 /** \brief Empty destructor */
247 ~MyPointRepresentation() override = default;
248
249 void
250 copyToFloatArray(const PointTarget& p, float* out) const override
251 {
252 out[0] = p.x;
253 out[1] = p.y;
254 out[2] = p.z;
255 }
256 };
257
258public:
259 PCL_MAKE_ALIGNED_OPERATOR_NEW
260};
