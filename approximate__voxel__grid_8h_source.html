44 /** \brief Helper functor structure for copying data between an Eigen::VectorXf and a PointT. */
45 template <typename PointT>
46 struct xNdCopyEigenPointFunctor
47 {
48 using Pod = typename traits::POD<PointT>::type;
49
50 xNdCopyEigenPointFunctor (const Eigen::VectorXf &p1, PointT &p2)
51 : p1_ (p1),
52 p2_ (reinterpret_cast<Pod&>(p2))
53 { }
54
55 template<typename Key> inline void operator() ()
56 {
57 //boost::fusion::at_key<Key> (p2_) = p1_[f_idx_++];
58 using T = typename pcl::traits::datatype<PointT, Key>::type;
59 std::uint8_t* data_ptr = reinterpret_cast<std::uint8_t*>(&p2_) + pcl::traits::offset<PointT, Key>::value;
60 *reinterpret_cast<T*>(data_ptr) = static_cast<T> (p1_[f_idx_++]);
61 }
62
63 private:
64 const Eigen::VectorXf &p1_;
65 Pod &p2_;
66 int f_idx_{0};
67 };

92 /** \brief ApproximateVoxelGrid assembles a local 3D grid over a given PointCloud, and downsamples + filters the data.
93 * Thus, this is similar to the \ref VoxelGrid filter.
94 * This class works best if points that are stored in memory next to each other (in the input point cloud), are also somewhat close in 3D euclidean space (this is for example usually the case for organized point clouds). If the points have no storage order (e.g. in synthetic, randomly generated point clouds), this class will give very poor results, and \ref VoxelGrid should be used instead.
95 * \sa VoxelGrid
96 * \author James Bowman, Radu B. Rusu
97 * \ingroup filters
98 */
99 template <typename PointT>
100 class ApproximateVoxelGrid: public Filter<PointT>
101 {
102 using Filter<PointT>::filter_name_;
103 using Filter<PointT>::getClassName;
104 using Filter<PointT>::input_;
105 using Filter<PointT>::indices_;
106
107 using PointCloud = typename Filter<PointT>::PointCloud;
108 using PointCloudPtr = typename PointCloud::Ptr;
109 using PointCloudConstPtr = typename PointCloud::ConstPtr;
110
111 private:
112 struct he
113 {
114 he () = default;
115 int ix{0}, iy{0}, iz{0};
116 int count{0};
117 Eigen::VectorXf centroid;
118 };
119
120 public:
121
122 using Ptr = shared_ptr<ApproximateVoxelGrid<PointT> >;
123 using ConstPtr = shared_ptr<const ApproximateVoxelGrid<PointT> >;
124
125
126 /** \brief Empty constructor. */
127 ApproximateVoxelGrid () :
128 pcl::Filter<PointT> (),
129 leaf_size_ (Eigen::Vector3f::Ones ()),
130 inverse_leaf_size_ (Eigen::Array3f::Ones ()),
131
132 history_ (new he[histsize_])
133 {
134 filter_name_ = "ApproximateVoxelGrid";
135 }
136
137 /** \brief Copy constructor.
138 * \param[in] src the approximate voxel grid to copy into this.
139 */
140 ApproximateVoxelGrid (const ApproximateVoxelGrid &src) :
141 pcl::Filter<PointT> (),
142 leaf_size_ (src.leaf_size_),
143 inverse_leaf_size_ (src.inverse_leaf_size_),
144 downsample_all_data_ (src.downsample_all_data_),
145 histsize_ (src.histsize_),
146 history_ ()
147 {
148 history_ = new he[histsize_];
149 for (std::size_t i = 0; i < histsize_; i++)
150 history_[i] = src.history_[i];
151 }
152
153
154 /** \brief Destructor.
155 */
156 ~ApproximateVoxelGrid () override
157 {
158 delete [] history_;
159 }
160
161
162 /** \brief Copy operator.
163 * \param[in] src the approximate voxel grid to copy into this.
164 */
165 inline ApproximateVoxelGrid&
166 operator = (const ApproximateVoxelGrid &src)
167 {
168 if (this == &src)
169 return *this;
170 leaf_size_ = src.leaf_size_;
171 inverse_leaf_size_ = src.inverse_leaf_size_;
172 downsample_all_data_ = src.downsample_all_data_;
173 histsize_ = src.histsize_;
174 history_ = new he[histsize_];
175 for (std::size_t i = 0; i < histsize_; i++)
176 history_[i] = src.history_[i];
177 return *this;
178 }

180 /** \brief Set the voxel grid leaf size.
181 * \param[in] leaf_size the voxel grid leaf size
182 */
183 inline void
184 setLeafSize (const Eigen::Vector3f &leaf_size)
185 {
186 leaf_size_ = leaf_size;
187 inverse_leaf_size_ = Eigen::Array3f::Ones () / leaf_size_.array ();
188 }
189
190 /** \brief Set the voxel grid leaf size.
191 * \param[in] lx the leaf size for X
192 * \param[in] ly the leaf size for Y
193 * \param[in] lz the leaf size for Z
194 */
195 inline void
196 setLeafSize (float lx, float ly, float lz)
197 {
198 setLeafSize (Eigen::Vector3f (lx, ly, lz));
199 }
200
201 /** \brief Get the voxel grid leaf size. */
202 inline Eigen::Vector3f
203 getLeafSize () const { return (leaf_size_); }
204
205 /** \brief Set to true if all fields need to be downsampled, or false if just XYZ.
206 * \param downsample the new value (true/false)
207 */
208 inline void
209 setDownsampleAllData (bool downsample) { downsample_all_data_ = downsample; }
210
211 /** \brief Get the state of the internal downsampling parameter (true if
212 * all fields need to be downsampled, false if just XYZ).
213 */
214 inline bool
215 getDownsampleAllData () const { return (downsample_all_data_); }
216
217 protected:
218 /** \brief The size of a leaf. */
219 Eigen::Vector3f leaf_size_;
220
221 /** \brief Compute 1/leaf_size_ to avoid division later */
222 Eigen::Array3f inverse_leaf_size_;
223
224 /** \brief Set to true if all fields need to be downsampled, or false if just XYZ. */
225 bool downsample_all_data_{true};
226
227 /** \brief history buffer size, power of 2 */
228 std::size_t histsize_{512};
229
230 /** \brief history buffer */

242 */

245 };
