45namespace pcl {
46namespace registration {
47/** \brief CorrespondenceRejectorOneToOne implements a correspondence
48 * rejection method based on eliminating duplicate match indices in
49 * the correspondences. Correspondences with the same match index are
50 * removed and only the one with smallest distance between query and
51 * match are kept. That is, considering match->query 1-m correspondences
52 * are removed leaving only 1-1 correspondences.
53 * \author Dirk Holz
54 * \ingroup registration
55 */
56class PCL_EXPORTS CorrespondenceRejectorOneToOne : public CorrespondenceRejector {
57 using CorrespondenceRejector::getClassName;
58 using CorrespondenceRejector::input_correspondences_;
59 using CorrespondenceRejector::rejection_name_;
60
61public:
62 using Ptr = shared_ptr<CorrespondenceRejectorOneToOne>;
63 using ConstPtr = shared_ptr<const CorrespondenceRejectorOneToOne>;
64
65 /** \brief Empty constructor. */
66 CorrespondenceRejectorOneToOne()
67 {
68 rejection_name_ = "CorrespondenceRejectorOneToOne";
69 }
70
71 /** \brief Get a list of valid correspondences after rejection from the original set
72 * of correspondences. \param[in] original_correspondences the set of initial
73 * correspondences given \param[out] remaining_correspondences the resultant filtered
74 * set of remaining correspondences
75 */
76 void
77 getRemainingCorrespondences(const pcl::Correspondences& original_correspondences,
78 pcl::Correspondences& remaining_correspondences) override;
79
80protected:
81 /** \brief Apply the rejection algorithm.
82 * \param[out] correspondences the set of resultant correspondences.
83 */
84 inline void
85 applyRejection(pcl::Correspondences& correspondences) override
86 {
87 getRemainingCorrespondences(*input_correspondences_, correspondences);
88 }
89};
